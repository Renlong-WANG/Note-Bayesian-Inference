\chapter{Bayesian Linear Regression}

\section{Basic Settings}

Consider a training dataset 
\[
    \mathcal{D} = \left\{(x_i,y_i)\right\}_{i=1}^{N},
\]
consisting of $N$ observations, where $\bm{x}_i \in \mathbb{R}^{p}$ denotes the $p$-dimensional feature vector associated with the $i$-th observation and $y_i \in \mathbb{R}$ denotes the corresponding scalar response. 

Define design matrix $\mathbf{X}$, target vector $\bm{y}$, and coefficient vector $\bm{w}$ as 
\[
    \mathbf{X} = \begin{bmatrix}
        \bm{x}_1^{\top} \\
        \vdots \\
        \bm{x}_N^{\top}
    \end{bmatrix} \in \mathbb{R}^{N \times p}, \qquad
    \bm{y} = \begin{bmatrix}
        {y}_1 \\
        \vdots \\
        {y}_N
    \end{bmatrix} \in \mathbb{R}^{N}, \qquad
    \bm{w} = \begin{bmatrix}
        {w}_1 \\
        \vdots \\
        {w}_p
    \end{bmatrix} \in \mathbb{R}^{p}.
\]

Under the linear regression model, each observation satisfies
\[
    y_i = \bm{w}^{\top} \bm{x}_i + \epsilon_i,
\]
where the noise terms are assumed to be independent and identically distributed as
\[
    \epsilon_i \overset{\mathrm{i.i.d.}}{\sim}\mathcal N(0,\sigma^2), \qquad i=1,\dots,N.
\]

Define the noise vector
\[
    \bm{\epsilon} = \begin{bmatrix}
        \epsilon_1 \\
        \vdots \\
        \epsilon_N
    \end{bmatrix}.
\]
It follows that
\[
    \bm\epsilon \sim \mathcal N(\bm 0,\sigma^2\mathbf I_N),
\]
where $\mathbf{I}_N$ denotes the $N\times N$ identity matrix.

For the entire dataset, the linear regression model can therefore be written as
\[
    \bm{y} = \mathbf{X}\bm{w} + \bm{\epsilon}.
\]
Consequently,
\[
    p(\bm{y} | \mathbf{X}, \bm{w}) \sim \mathcal{N}(\mathbf{X}\bm{w}, \sigma^2 \mathbf{I}_N).
\]

Bayesian method includes two basic steps: inference and prediction, where inference is to estimate posterior $\bm{w}$, while prediction is to predict ${y}^{*}$ given $\bm{x}^{*}$.

\section{Inference}


\subsection{Model Establishment}

In Bayesian linear regression, the coefficient vector $\bm{w}$ is treated as a random variable and assigned a prior distribution. We assume that the prior distribution of $\bm{w}$ does not depend on $\mathbf{X}$, i.e., $w\perp X$, and thus $p(\bm w\mid\mathbf X)=p(\bm w)$.

Given the training dataset, Bayesian formula is decomposed as: 
\[
    p(\bm{w} | \mathcal{D}) = p(\bm{w} | \mathbf{X}, \bm{y}) = \frac{p(\bm{w}, \mathbf{X}, \bm{y})}{\textcolor{red}{p(\mathbf{X}, \bm{y})}} = \frac{\textcolor{red}{p(\bm{w}, \mathbf{X}, \bm{y})}}{p(\bm{y}|\mathbf{X}) \textcolor{red}{p(\mathbf{X})} } = \frac{\textcolor{red}{p(\bm{w}, \bm{y} | \mathbf{X})}}{p(\bm{y}|\mathbf{X})} = \frac{p(\bm{y} | \mathbf{X}, \bm{w}) \textcolor{red}{p(\bm{w} | \mathbf{X})}}{p(\bm{y}|\mathbf{X})} = \frac{p(\bm{y} | \mathbf{X}, \bm{w}) p(\bm{w})}{p(\bm{y}|\mathbf{X})},
\]
where the fifth equality follows from
\[
    p(\bm{y} | \mathbf{X}, \bm{w}) = \frac{p(\bm{w}, \mathbf{X}, \bm{y})}{p(\bm{w}, \mathbf{X})} = \frac{p(\bm{w}, \mathbf{X}, \bm{y})}{ p(\bm{w} | \mathbf{X}) p(\mathbf{X})} = \frac{p(\bm{w}, \bm{y} | \mathbf{X})}{p(\bm{w} | \mathbf{X})},
\]
and the last equality follows from independency of prior coefficient and design matrix. 

The denominator 
\[
    p(\bm{y}\mid\mathbf{X}) = \int p(\bm{y}\mid\mathbf{X},\bm{w}) p(\bm{w})\mathrm{d}\bm{w}
\]  
is the marginal likelihood. Since it does not depend on $\bm{w}$, the posterior distribution can be expressed up to a proportionality constant as 
\[
    p(\bm{w} | \mathbf{X}, \bm{y}) \propto p(\bm{y} | \mathbf{X}, \bm{w}) p(\bm{w}),
\]
where $p(\bm{y} | \mathbf{X}, \bm{w})$ is a likelihood function and $p(\bm{w})$ is a prior function.

Suppose that a zero-mean Gaussian prior is imposed on the coefficient vector:
\[
    p(\bm{w}) = \mathcal{N}\left(\bm{0}, \mathbf{\Sigma}_{p}\right),
\]
where $\mathbf{\Sigma}_p\in\mathbb{R}^{p\times p}$ denotes the prior covariance matrix.

Since both the likelihood and the prior are Gaussian, their conjugacy implies that the posterior distribution of $\bm{w}$ is also Gaussian:
\[
    p(\bm{w} | \mathbf{X}, \bm{y}) = \mathcal{N}(\bm{\mu}_{\bm{w}}, \mathbf{\Sigma}_{\bm{w}})
\]
Thus,
\[
    p(\bm{w} | \mathbf{X}, \bm{y}) \propto \mathcal{N}(\mathbf{X}\bm{w}, \sigma^2 \mathbf{I}_N) \mathcal{N}\left(\bm{0}, \mathbf{\Sigma}_{p}\right).
\]

The remaining task is therefore to derive the posterior mean $\bm{\mu}_w$ and posterior covariance $\mathbf{\Sigma}_w$.

\subsection{Model Solution}

To derive the posterior distribution, we first write the product of the likelihood and the prior explicitly:
\[
    \mathcal{N}(\mathbf{X}\bm{w}, \sigma \mathbf{I}_N) \mathcal{N}\left(\bm{0}, \mathbf{\Sigma}_{p}\right) = 
    \frac{1}{(2\pi)^{N/2} \sigma^{N}}\exp\left(-\frac{1}{2\sigma^{2}}  (\bm{y} - \mathbf{X}\bm{w})^{\top} (\bm{y} - \mathbf{X}\bm{w}) \right) \frac{1}{(2\pi)^{p/2} |\mathbf{\Sigma}_{p}|^{1/2}}\exp\left(-\frac{1}{2} (\bm{w})^{\top} \mathbf{\Sigma}_{p}^{-1} \bm{w} \right).
\]

Thus, 
\[
\begin{aligned}
    p(\bm{w} | \mathbf{X}, \bm{y}) \propto & \exp\left(-\frac{1}{2\sigma^{2}}  (\bm{y} - \mathbf{X}\bm{w})^{\top} (\bm{y} - \mathbf{X}\bm{w}) -\frac{1}{2} (\bm{w})^{\top} \mathbf{\Sigma}_{p}^{-1} \bm{w}\right) \\
    & = \exp\left(-\frac{1}{2\sigma^{2}} \left(\bm{y}^{\top} \bm{y} \textcolor{green}{- \bm{y}^{\top} \mathbf{X}\bm{w}} \textcolor{green}{- \bm{w}^{\top} \mathbf{X}^{\top} \bm{y}} \textcolor{blue}{+ \bm{w}^{\top} \mathbf{X}^{\top} \mathbf{X} \bm{w}}\right) - \frac{1}{2} \textcolor{blue}{(\bm{w})^{\top} \mathbf{\Sigma}_{p}^{-1} \bm{w}} \right) \\
    & = \exp\left( -\frac{1}{2\sigma^{2}} \bm{y}^{\top} \bm{y} + \frac{1}{\sigma^{2}} \bm{w}^{\top} \mathbf{X}^{\top} \bm{y} - \frac{1}{2} \bm{w}^{\top} \left(\frac{1}{\sigma^{2}} \mathbf{X}^{\top} \mathbf{X} + \mathbf{\Sigma}_{p}^{-1}\right) \bm{w} \right).
\end{aligned}
\]

Comparing this expression with the kernel of a multivariate Gaussian distribution,
\[
    \exp\left(-\frac{1}{2} (\bm{w} - \bm{\mu}_{\bm{w}})^{\top} \mathbf{\Sigma}_{\bm{w}}^{-1} (\bm{w} - \bm{\mu}_{\bm{w}}) \right) = 
    \exp\left( -\frac{1}{2} \bm{w}^{\top} \mathbf{\Sigma}_{\bm{w}}^{-1} \bm{w} + \bm{w}^{\top} \mathbf{\Sigma}_{\bm{w}}^{-1} \bm{\mu}_{\bm{w}} - \frac{1}{2} \bm{\mu}_{\bm{w}}^{\top} \mathbf{\Sigma}_{\bm{w}}^{-1} \bm{\mu}_{\bm{w}} \right),
\]
we obtain
\[
    \mathbf{\Sigma}_{\bm{w}}^{-1} = \frac{1}{\sigma^{2}} \mathbf{X}^{\top} \mathbf{X} + \mathbf{\Sigma}_{p}^{-1} = \mathbf{A}.
\]

Matching the linear terms gives
\[
    \bm{w}^{\top} \mathbf{\Sigma}_{\bm{w}}^{-1} \bm{\mu}_{\bm{w}} = \frac{1}{\sigma^{2}} \bm{w}^{\top} \mathbf{X}^{\top} \bm{y} \Leftrightarrow \mathbf{A} \bm{\mu}_{\bm{w}} = \frac{1}{\sigma^{2}} \mathbf{X}^{\top} \bm{y},
\]
and therefore
\[
    \bm{\mu}_{\bm{w}} = \frac{1}{\sigma^{2}} \mathbf{A}^{-1} \mathbf{X}^{\top} \bm{y}.
\]

Hence, the posterior distribution has the closed-form solution
\[
    p(\bm{w} | \mathbf{X}, \bm{y}) \sim \mathcal{N}(\bm{\mu}_{\bm{w}}, \mathbf{\Sigma}_{\bm{w}}),
\]
where 
\[
    \mathbf{\Sigma}_{\bm{w}}^{-1} = \frac{1}{\sigma^{2}} \mathbf{X}^{\top} \mathbf{X} + \mathbf{\Sigma}_{p}^{-1}, \qquad \bm{\mu}_{\bm{w}} = \frac{1}{\sigma^{2}} \mathbf{\Sigma}_{\bm{w}} \mathbf{X}^{\top} \bm{y}.
\]


\section{Prediction}

After obtaining the posterior distribution of the coefficient vector, Bayesian linear regression can be used to predict the responses associated with new observations.

Let $\mathbf{X}^{*}\in\mathbb{R}^{N^{*}\times p}$ denote the design matrix for $N^{*}$ new observations, and let $\bm{y}^{*}\in\mathbb{R}^{N^{*}}$ denote the corresponding target vector to be predicted.

Given the posterior distribution, the posterior predictive distribution is obtained by marginalizing over the uncertainty in $\bm{w}$:
\[
    p(\bm{y}^{*}\mid\mathbf{X}^{*},\mathcal{D}) = \int p(\bm{y}^{*}\mid\mathbf{X}^{*},\bm{w})p(\bm{w}\mid\mathcal{D})\mathrm{d}\bm{w}.
\]

Thus, we have 
\[
    p(\bm{y}^{*}\mid\mathbf{X}^{*},\mathcal{D}) = \mathcal{N}\left(\mathbf{X}^{*}\bm{\mu}_{\bm{w}}, \mathbf{X}^{*}\mathbf{\Sigma}_{w}(\mathbf{X}^{*})^{\top} + \sigma^{2}\mathbf{I}_{N^{*}} \right).
\]

Therefore, the predictive mean and covariance are
\[
    \mathbb{E}\left[\bm{y}^{*}\mid\mathbf{X}^{*},\mathcal{D}\right]=\mathbf{X}^{*}\bm{\mu}_{w},
\]
and
\[
    \operatorname{Cov}\left(\bm{y}^{*}\mid\mathbf{X}^{*},\mathcal{D}\right)=\underbrace{\mathbf{X}^{*}\mathbf{\Sigma}_{w}(\mathbf{X}^{*})^{\top}}_{\text{parameter uncertainty}} + \underbrace{\sigma^{2}\mathbf{I}_{N^{*}}}_{\text{observation noise}}.
\]
